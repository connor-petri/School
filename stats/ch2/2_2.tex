\subsection{Probability}
\hrulefill

\paragraph*{Definition}
Probability is a function that maps the sample space S to $\mathbb{R}$. The sample space is the 
\textbf{domain} and the real numbers are the \textbf{codomain}.

\subsubsection*{Axioms of Probability}
\begin{enumerate}
    \item Non-negativity: For any event A, $P(A) \geq 0$.
    \item Normalization: $P(S) = 1$, where S is the sample space.
    \item Additivity: For any two mutually exclusive events A and B, $P(A_1 \cup A_2 \cup \cdots \cup A_n) = \sum_{i=1}^{n} P(A_i)$.
\end{enumerate}

\begin{equation*}
    P(\cup_{i=1}^{n} A_i) = \sum_{i=1}^{n} P(A_i)
\end{equation*}

\paragraph*{Not axioms but useful observations}
\begin{enumerate}
    \item $P(A) < 1$ for any event A.
    \item $P(A^c) = 1 - P(A)$.
    \item $P(A \cap B) = P(A) + P(B) - P(A \cup B)$ for any two events A and B.
\end{enumerate}

\subsubsection*{Proving Observations}

\paragraph*{Observation 1: $P(\emptyset) = 0$}
\begin{align*}
    P(\emptyset) &= 0 \\
    \emptyset \cup S &= S \\
    \emptyset \cap S &= \emptyset \\
    P(\emptyset \cup S) &= P(S) \\
    P(\emptyset) + P(S) &= P(S) \quad \text{(by axiom 3)}
\end{align*}
\noindent
$\emptyset$ and $S$ are mutually exclusive.\\
Therefore, $P(\emptyset) = 0$ by algebra.

\paragraph*{Observation 2: $P(A^c) = 1 - P(A)$}
\begin{align*}
    S &= A \cup A^c \\
    \emptyset &= A \cap A^c \\
    P(A^c \cup A) &= P(S) \\
    P(A^c) + P(A) &= P(S) \quad \text{(by axiom 3)} \\
    P(A^c) + P(A) &= 1 \\
    P(A^c) &= 1 - P(A)
\end{align*}

\paragraph*{Observation 3: $P(A \cup B) = P(A) + P(B)$ if $A \cap B = \emptyset$}
\begin{align*}
    P\left(\bigcup_{i=1}^\infty A_i\right) &= \sum_{i=1}^\infty P(A_i) \quad \text{for mutually exclusive } A_i \\
    \text{Let } A_1 = A,\, A_2 = B,\, A_3 = A_4 = \emptyset \\
    P(A \cup B \cup \emptyset \cup \emptyset) &= P(A) + P(B) + 0 \\
    P(A \cup B) &= P(A) + P(B) \quad \text{(by axiom 3)}
\end{align*}

\paragraph*{Identities for the next proof}
\begin{align*}
    P(A) &= P(A \cap B') + P(A \cap B) \\
    P(B) &= P(A' \cap B) + P(A \cap B) \\
    P(A \cup B) &= P(A \cap B') + P(A \cap B) + P(A' \cap B) \\
    1 &= P(A \cap B') + P(A \cap B) + P(A' \cap B) + P(A' \cap B')
\end{align*}

\paragraph*{Observation 4: $P(A \cup B) = P(A) + P(B) - P(A \cap B)$}
\begin{align*}
    P(A \cup B) &= P(A \cap B') + P(A \cap B) + P(A' \cap B) \\
    &= P(A) + P(A' \cap B) \\
    &= P(A) + [P(B) - P(A \cap B)] \\
    &= P(A) + P(B) - P(A \cap B)
\end{align*}

\subsubsection*{Other Propositions}

\paragraph*{$A \subseteq B \implies P(A) \leq P(B)$}
\begin{align*}
    B &= (A \cap B) \cup (A' \cap B) \\
    A \cap B &= A \\
    B &= A \cup (A' \cap B) \\
    P(B) &= P(A) + P(A' \cap B) \quad \text{(by axiom 3)} \\
    P(B) &\geq P(A) \quad \text{since } P(A' \cap B) \geq 0 \text{ (by axiom 1)} \\
    \text{For any } A,\, A &\subseteq S \\
    P(A) &\leq P(S) \\
    \therefore P(A) &< 1
\end{align*}

\paragraph*{Example}
If $P(C) = .46, P(D) = .47, P(C \cup D) = .62$, find the following:
\begin{align*}
    P(C') &= 1 - P(C) = 1 - .46 = .54 \\
    P(C \cap D) &= P(C) + P(D) - P(C \cup D) = .46 + .47 - .62 = .31 \\
    P(C \cap D') &= P(C) - P(C \cap D) = .46 - .31 = .15 \\
    P(C' \cup D') &= 1 - P(C \cup D) = 1 - .62 = .38 \\
\end{align*}
