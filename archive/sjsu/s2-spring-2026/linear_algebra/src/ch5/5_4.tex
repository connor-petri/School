\subsection*{Rank of a Matrix}
\paragraph*{Definition}
The \textit{rank} of a matrix $A$, denoted $\mathrm{rank}(A)$, is the maximum number of linearly independent rows or columns in $A$.
Previously we have seen that the rank of a matrix is equal to the number of pivots in its row echelon form, which is also equal to the number of nonzero rows in its reduced row echelon form. 
The rank of a matrix can also be defined as the dimension of the column space or the row space of the matrix.

\paragraph*{Definition}
The \emph{column space} of a matrix $A$, denoted $\mathrm{Col}(A) = \text{span}\{\vec{a}_1, \vec{a}_2, \ldots, \vec{a}_n\} = \text{Im}(A)$, is the set of all linear combinations of the columns of $A$. 
The column space is a subspace of $\mathbb{R}^n$ if $A$ is an $m \times n$ matrix.

The \emph{row space} of a matrix $A$, denoted $\mathrm{Row}(A)$, is the set of all linear combinations of the rows of $A$.
The row space is a subspace of $\mathbb{R}^m$ if $A$ is an $m \times n$ matrix.

\paragraph*{Example}
Let $A = \begin{bmatrix}
    1 & -3 & 5 \\ 2 & 0 & 22
\end{bmatrix}$. Find spanning sets for $\mathrm{Col}(A)$ and $\mathrm{Row}(A)$, and determine $\mathrm{rank}(A)$.
\begin{align*}
\mathrm{Col}(A) & = \text{span}\left\{\begin{bmatrix}
    1 \\ 2
\end{bmatrix}, \begin{bmatrix}    -3 \\ 0
\end{bmatrix}, \begin{bmatrix}    5 \\ 22
\end{bmatrix}\right\} \\
\mathrm{Row}(A) & = \text{span}\left\{\begin{bmatrix}
    1 & -3 & 5
\end{bmatrix}, \begin{bmatrix}    
    2 & 0 & 22
\end{bmatrix}\right\} \\
\mathrm{rank}(A) & = 2 \\
\therefore \quad & \text{ a basis for Col(A) is } B = \left\{\begin{bmatrix}
    1 \\ 2
\end{bmatrix}, \begin{bmatrix}    
    -3 \\ 0
\end{bmatrix}\right\}\\
B &= \left\{ \begin{bmatrix}
    1 & -3 & 5
\end{bmatrix}, \begin{bmatrix}    
    2 & 0 & 22
\end{bmatrix}\right\} \text{ is a basis for Row(A) because the two rows are linearly independent.}
\end{align*}

\paragraph*{Properties of Rank}
\begin{itemize}
    \item $\dim(\mathbb{R}^n) = n$.
    \item $0 \leq \dim{\text{subspace}{\mathbb{R}^n}} \leq n$
\end{itemize}

\paragraph*{Theorem}
Let $A, B \in \mathbb{R}^{m \times n}$, then
\begin{enumerate}
    \item If $A \rightarrow B$ by elementary row operations, then $\text{Row}(A) = \text{Row}(B)$.
    \item If $A \rightarrow B$ by elementary \emph{column} operations, then $\text{Col}(A) = \text{Col}(B)$.
    \item $\text{Row}(A) = \text{Row}(\text{RREF}(A)) = \text{Row(\text{REF}(A))}$.
    \item $\text{Col}(A) \neq \text{Col}(\text{RREF}(A)) \neq \text{Col}(\text{REF}(A))$.
    \item Substituting a vector in a span with a linear combination of the vectors in the span does not change the span. (Closed under addition and scalar multiplication)
\end{enumerate}

\paragraph*{Theorem}
Let $A \in \mathbb{R}^{m \times n}$ and $A \rightarrow REF(A)$, then
\begin{enumerate}
    \item A basis of $\text{Row}(A)$ is given by the nonzero rows of $REF(A)$.
    \item A basis of $\text{Col}(A)$ is given by the columns of $A$ that \emph{correspond} to the pivot columns of $REF(A)$.
\end{enumerate}

\paragraph*{Example}
Let $A = \begin{bmatrix}
    -3 & 0 & -3 & -3 \\ 1 & 2 & 3 & -1 \\ 2 & -1 & 1 & 3
\end{bmatrix}$ and $REF(A) = \begin{bmatrix}
    1 & 0 & 1 & 1 \\ 0 & 1 & 1 & -1 \\ 0 & 0 & 0 & 0
\end{bmatrix}$. Find a basis for $\text{Row}(A)$ and $\text{Col}(A)$.
\begin{align*}
    B_\text{Col} &= \left\{\begin{bmatrix}
        -3 \\ 1 \\ 2
    \end{bmatrix}, \begin{bmatrix}
        0 \\ 2 \\ -1
    \end{bmatrix}\right\}\\
    \implies & \text{Col}(A) = \text{span}\left\{\begin{bmatrix}
        -3 \\ 1 \\ 2
    \end{bmatrix}, \begin{bmatrix}        
        0 \\ 2 \\ -1
    \end{bmatrix}\right\}\\
    B_\text{Row} &= \left\{\begin{bmatrix}
        1 & 0 & 1 & 1
    \end{bmatrix}, \begin{bmatrix}
        0 & 1 & 1 & -1
    \end{bmatrix}\right\} \\
    \implies & \text{Row}(A) = \text{span}\left\{\begin{bmatrix}
        1 & 0 & 1 & 1
    \end{bmatrix}, \begin{bmatrix}
        0 & 1 & 1 & -1
    \end{bmatrix}\right\}
\end{align*}

\paragraph*{Example}
Find a basis of $U = \left\{\begin{bmatrix}
    1 \\ 2 \\ 1
\end{bmatrix}, \begin{bmatrix}
    1 \\ 4 \\ 5
\end{bmatrix}, \begin{bmatrix}
    2 \\ 1 \\ -4
\end{bmatrix}, \begin{bmatrix}
    3 \\ 0 \\ -9
\end{bmatrix}\right\}$.
\begin{align*}
    A &= \begin{bmatrix}
        1 & 1 & 2 \\ 2 & 4 & 1 & 0 \\ 1 & 5 & -4 & -9
    \end{bmatrix} \xmapsto{REF} \begin{bmatrix}
        1 & 1 & 2 & 3 \\ 0 & 1 & -\frac{3}{2} & -3 \\ 0 & 0 & 0 & 0
    \end{bmatrix}\\
    \implies & B_U = \left\{\begin{bmatrix}
        1 \\ 2 \\ 1
    \end{bmatrix}, \begin{bmatrix}
        1 \\ 4 \\ 5
    \end{bmatrix}\right\} \implies \dim{U} = 2
\end{align*}

\paragraph*{Theorem}
Let $A \in \mathbb{R}^{m \times n}$ with $\mathrm{rank}(A) = r$, then 
\begin{enumerate}
    \item $A$ has $r$ independent rows and $r$ independent columns.
    \item $\mathrm{dim}(\mathrm{Row}(A)) = \mathrm{dim}(\mathrm{Col}(A)) = r$.
\end{enumerate}

\paragraph*{Useful Facts}
Let $A \in \mathbb{R}^{m \times n}$, then
\begin{enumerate}
    \item $\mathrm{rank}(A) = \mathrm{rank}(A^T)$.
    \item $\mathrm{rank}(A) \leq \min\{m, n\}$.
    \item Let $V \in \mathbb{R}^{m \times m}$ and $W \in \mathbb{R}^{n \times n}$ be invertible matrices, then $\mathrm{rank}(A) = \mathrm{rank}(VA) = \mathrm{rank}(AW)$.
    \item For non-invertible matrices $V$ and $W$, $\mathrm{rank}(A) \leq \mathrm{rank}(VA)$ and $\mathrm{rank}(A) \leq \mathrm{rank}(AW)$.
\end{enumerate}

\subsubsection*{Nullity of a Matrix}
\paragraph*{Definition}
The \emph{nullity} of a matrix $A$, denoted $\mathrm{nullity}(A)$, is the dimension of the null space of $A$, 
which is the set of all solutions to the homogeneous equation $A\vec{x} = \vec{0}$.

\paragraph*{Rank-Nullity Theorem}
If $A \in \mathbb{R}^{m \times n}$, then $\mathrm{rank}(A) + \mathrm{nullity}(A) = n \implies \mathrm{nullity}(A) = n - \mathrm{rank}(A)$.
This means the number of free variables in the solution to $A\vec{x} = \vec{0}$ is equal to the nullity of $A$.
