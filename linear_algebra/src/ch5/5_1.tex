\subsection{Subspaces and Spanning Sets}
\subsubsection{Subspaces}
\paragraph*{Definition} A set $U \in \mathbb{R}^n$ is a \emph{subspace} of $\mathbb{R}^n$ if and only if:
\begin{enumerate}
    \item $\vec{0} \in U$, or $U$ cannot be empty
    \item $\vec{x}, \vec{y} \in U \implies \vec{x} + \vec{y} \in U$, or $U$ is closed under vector addition
    \item $\vec{x} \in U \implies \forall k \in \mathbb{R}, k\vec{x} \in U$, or $U$ is closed under scalar multiplication
\end{enumerate}

\paragraph*{Properties}
\begin{enumerate}
    \item $U = \{ \vec{0} \} \subseteq \mathbb{R}^n$ is a subspace of $\mathbb{R}^n$
    \item $\mathbb{R}^n$ is a subspace of $\mathbb{R}^n$
    \item Any subspace of $\mathbb{R}^n$ other than $\{ \vec{0} \}$ and $\mathbb{R}^n$ is a \emph{proper subspace} of $\mathbb{R}^n$
    \item Any line through the origin in $\mathbb{R}^2$ is a proper subspace of $\mathbb{R}^2$.\\
    \begin{itemize}
        \item $\ell = \begin{bmatrix}
            x \\
            mx
        \end{bmatrix}$ for some $x \in \mathbb{R}$
        \item $\vec{0} \in \ell$
        \item $\begin{bmatrix}
            x_1 \\
            mx_1
        \end{bmatrix} + \begin{bmatrix}
            x_2 \\
            mx_2
        \end{bmatrix} = \begin{bmatrix}
            x_1 + x_2 \\
            m(x_1 + x_2)
        \end{bmatrix} \in \ell$
        \item $k \begin{bmatrix}
            x \\
            mx
        \end{bmatrix} = \begin{bmatrix}
            kx \\
            kmx
        \end{bmatrix} \in \ell$
        \item $\therefore, \ell$ is a subspace of $\mathbb{R}^2$
    \end{itemize}
    \item Lines are the \emph{only} proper subspaces of $\mathbb{R}^2$
    \item Lines and planes through the origin are the \emph{only} proper subspaces of $\mathbb{R}^3$
\end{enumerate}

Is $M = \{\vec{n} \cdot \vec{m} = 0 | \vec{p} \in \mathbb{R}^3\}$, is $M$ a subspace of $\mathbb{R}^3$?\\
\begin{enumerate}
    \item $\vec{n} \cdot \vec{0} = 0 \implies \vec{0} \in M$
    \item $\vec{u}, \vec{v} \in M \implies \vec{n} \cdot (\vec{u} + \vec{v}) = \vec{n} \cdot \vec{u} + \vec{n} \cdot \vec{v} = 0 + 0 = 0 \implies \vec{u} + \vec{v} \in M$
    \item $\vec{v} \in M \implies \vec{n} \cdot (k\vec{v}) = k(\vec{n} \cdot \vec{v}) = k \cdot 0 = 0 \implies k\vec{v} \in M$
\end{enumerate}

\subsubsection*{Null Space and Image Space}
\paragraph*{Definition} Let $A$ be an $m \times n$ matrix. Then
\begin{itemize}
    \item $\text{Null}(A) = \{\vec{x} \in \mathbb{R}^n | A\vec{x} = \vec{0}\}$ is the \emph{null space} of $A$. This is the set of all solutions to the homogeneous system $A\vec{x} = \vec{0}$, and is a subspace of $\mathbb{R}^n$.
    \item $\text{Image}(A) = \{A\vec{x} | \vec{x} \in \mathbb{R}^n\} \subseteq \mathbb{R}^m$ is the \emph{image space} of $A$. This is the set of all vectors that can be expressed as a linear combination of the columns of $A$, and is a subspace of $\mathbb{R}^m$. Also known as the \emph{column space} of $A$.
    \item The image set can also be described as all vectors $\vec{b} \in \mathbb{R}^m$ such that $A\vec{x} = \vec{b}$ is consistent.
\end{itemize}

\paragraph*{Eigenspace}
The \emph{eigenspace} of a matrix $A$ corresponding to an eigenvalue $\lambda$ is the null space of $A - \lambda I$. It is the set of all eigenvectors corresponding to $\lambda$, along with the zero vector. It is a subspace of $\mathbb{R}^n$.


\paragraph*{Example}
Let $A = \begin{bmatrix}
    1 & 3 \\
    0 & 2
\end{bmatrix}$. This implies $\lambda_1 = 1$ and $\lambda_2 = 2$.
\begin{enumerate}
    \item Find $E_{\lambda_1}(A)$
    \begin{align*}
    E_{\lambda_1}(A) &= \text{Null}(A - \lambda_1 I) \\
    &= \text{Null}\left(\begin{bmatrix}
    0 & 3 \\
    0 & 1
\end{bmatrix} \right)
    \end{align*}
\end{enumerate}
